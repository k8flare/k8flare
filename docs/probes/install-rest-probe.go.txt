//go:build js && wasm

package main

import (
	"context"
	"fmt"

	"github.com/emicklei/go-restful/v3"
	corev1 "k8s.io/api/core/v1"
	"k8s.io/apimachinery/pkg/api/validation/path"
	"k8s.io/apimachinery/pkg/runtime"
	"k8s.io/apimachinery/pkg/runtime/schema"
	"k8s.io/apimachinery/pkg/runtime/serializer"
	"k8s.io/apimachinery/pkg/util/validation/field"
	"k8s.io/apimachinery/pkg/util/managedfields"
	"k8s.io/apiserver/pkg/endpoints"
	genericregistry "k8s.io/apiserver/pkg/registry/generic/registry"
	"k8s.io/apiserver/pkg/registry/rest"
	"k8s.io/apiserver/pkg/storage/names"
	"k8s.io/client-go/kubernetes/scheme"
)

type strat struct {
	runtime.ObjectTyper
	names.NameGenerator
}

func (strat) NamespaceScoped() bool                                             { return true }
func (strat) PrepareForCreate(context.Context, runtime.Object)                  {}
func (strat) PrepareForUpdate(context.Context, runtime.Object, runtime.Object)  {}
func (strat) Validate(context.Context, runtime.Object) field.ErrorList          { return nil }
func (strat) WarningsOnCreate(context.Context, runtime.Object) []string         { return nil }
func (strat) Canonicalize(runtime.Object)                                       {}
func (strat) AllowCreateOnUpdate() bool                                         { return false }
func (strat) AllowUnconditionalUpdate() bool                                    { return true }
func (strat) ValidateUpdate(context.Context, runtime.Object, runtime.Object) field.ErrorList {
	return nil
}
func (strat) WarningsOnUpdate(context.Context, runtime.Object, runtime.Object) []string {
	return nil
}
func (strat) GenerateName(base string) string { return base }
func (strat) ObjectKinds(o runtime.Object) ([]schema.GroupVersionKind, bool, error) {
	return scheme.Scheme.ObjectKinds(o)
}
func (strat) Recognizes(schema.GroupVersionKind) bool { return true }

func main() {
	_ = path.ValidatePathSegmentName
	s := scheme.Scheme
	cf := serializer.NewCodecFactory(s)
	st := strat{ObjectTyper: s, NameGenerator: names.SimpleNameGenerator}
	store := &genericregistry.Store{
		NewFunc:                   func() runtime.Object { return &corev1.ConfigMap{} },
		NewListFunc:               func() runtime.Object { return &corev1.ConfigMapList{} },
		DefaultQualifiedResource:  schema.GroupResource{Resource: "configmaps"},
		SingularQualifiedResource: schema.GroupResource{Resource: "configmap"},
		CreateStrategy:            st,
		UpdateStrategy:            st,
		DeleteStrategy:            st,
		TableConvertor:            rest.NewDefaultTableConvertor(schema.GroupResource{Resource: "configmaps"}),
	}
	g := &endpoints.APIGroupVersion{
		Storage:                    map[string]rest.Storage{"configmaps": store},
		Root:                       "/api",
		GroupVersion:               schema.GroupVersion{Version: "v1"},
		Creater:                    s,
		Convertor:                  s,
		Typer:                      s,
		Serializer:                 cf,
		ParameterCodec:             runtime.NewParameterCodec(s),
		EquivalentResourceRegistry: runtime.NewEquivalentResourceRegistry(),
		TypeConverter:              managedfields.NewDeducedTypeConverter(),
	}
	c := restful.NewContainer()
	res, _, err := g.InstallREST(c)
	fmt.Println("InstallREST err:", err)
	fmt.Println("registered resources:", len(res), " routes:", len(c.RegisteredWebServices()))
	for _, ws := range c.RegisteredWebServices() {
		for _, r := range ws.Routes() {
			fmt.Println("  ", r.Method, r.Path)
		}
	}
}
